1) COPIAR DESDE EL PUENTE

   # Si no existe, probar: Y:\ApolloSupport

   Copy-Item "$puente\backend\main.py"      "C:\ApolloSupport\backend\main.py" -Force
   Copy-Item "$puente\backend\models.py"    "C:\ApolloSupport\backend\models.py" -Force
   Copy-Item "$puente\backend\schemas.py"   "C:\ApolloSupport\backend\schemas.py" -Force

2) REINICIAR BACKEND

   nssm restart ApolloBackend
   # Si nssm no está en PATH:
   #   C:\ApolloSupport\backend\venv\Scripts\nssm.exe restart ApolloBackend

   #   Restart-Service ApolloBackend -Force
   #   sc stop ApolloBackend && timeout /t 3 && sc start ApolloBackend

3) VERIFICAR

   sc query ApolloBackend
   # Debe estar RUNNING

   curl -s http://127.0.0.1:8001/api/version
